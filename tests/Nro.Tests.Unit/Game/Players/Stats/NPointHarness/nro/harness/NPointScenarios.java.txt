    static PS base() {
        PS p = new PS("player");
        p.set("p.id", 1L).set("p.isPlayer", true).set("p.name", "Tester")

    // ---------------------------------------------------------------- option tables (item table and card table)

    static void optionTables() {
        for (int id : OPTION_IDS) {
            for (int param : new int[]{7, 1250}) {
                add(new Case("option/" + id + "/p" + param, base().body(0, item(100, 0, o(id, param)))).op("calPoint"));
            }
            add(new Case("card-option/" + id, base().card(1, 0, oc(id, 9, 0))).op("calPoint"));
        }
        // the same option on five items: set counters and additive options accumulate
        for (int id : new int[]{34, 35, 36}) {
            for (int count : new int[]{4, 5, 6}) {
                PS p = base();
                for (int i = 0; i < count; i++) {
                    p.body(i, item(200 + i, 0, o(id, 1)));
                }
                add(new Case("set-seal/" + id + "/x" + count, p).op("calPoint").op("getDameAttack", "isMob", false, "seed", 5L));
            }
        }
    }

    /** Several flags at once, to check the ORDER of the multiplications (doubling before/after a percent, 2.2x cut to long). */
    static void itemTimeCombos() {
        Random r = new Random(20261103);
        for (int k = 0; k < 14; k++) {
            PS p = base();
            p.set("npoint.hpg", 1_000_000L + r.nextInt(900_000)).set("npoint.mpg", 1_000_000L + r.nextInt(900_000)).set("npoint.dameg", 50_000L + r.nextInt(90_000));
            int n = 3 + r.nextInt(10);
            StringBuilder names = new StringBuilder();
            for (int i = 0; i < n; i++) {
                String f = IT_FLAGS[r.nextInt(IT_FLAGS.length)];
                p.set("it." + f, true);
                names.append(f).append(' ');
            }
            calc("itemtime-combo/" + k, p);
        }
        PS all = base();
        all.set("npoint.hpg", 3_000_000L).set("npoint.mpg", 2_000_000L).set("npoint.dameg", 90_000L);
        for (String f : IT_FLAGS) {
            all.set("it." + f, true);
        }
        calc("itemtime-all-on", all);
        PS allBig = base();
        allBig.set("npoint.hpg", 2_000_000_000_000_000L).set("npoint.mpg", 1_500_000_000_000_000L).set("npoint.dameg", 900_000_000_000_000L);
        for (String f : IT_FLAGS) {
            allBig.set("it." + f, true);
        }
        calc("itemtime-all-on-big", allBig);
    }

    static void mapMatrix() {
        String[] flags = {"isMapCold", "isMapChristMasEvent", "isMap5000NamTruoc", "isMapCereal", "isMapVohinh", "isMapBlackBallWar"};
        for (String f : flags) {
            for (String who : new String[]{"player", "boss", "bot", "bot-event", "xayda", "xayda-boss", "khonglanh"}) {
                PS p = base().set("zone." + f, true).set("ek.xHPKI", 3).set("ek.xDame", 4);
                switch (who) {
                    case "boss":
                        p.set("p.isBoss", true);
                        break;
                    case "bot":
                        p.set("p.isBot", true);
                        break;
                    case "bot-event":
                        p.set("p.isBot_Event", true);
                        break;
                    case "xayda":
                        p.set("p.gender", 2);
                        break;
                    case "xayda-boss":
                        p.set("p.gender", 2).set("p.isBoss", true);
                        break;
                    case "khonglanh":
                        p.body(0, item(100, 0, o(106, 1)));
                        break;
                    default:
                        break;
                }
                add(new Case("map/" + f + "/" + who, p).op("calPoint").op("getDameAttack", "isMob", false, "seed", 9L));
            }
        }
        calc("map/none", base());
        calc("map/type3-zone", base().set("zone.type", 3));
    }

    static void petFusion() {
        int[] fusions = {0, 4, 6, 8, 10, 12, 14};
        int[] types = {0, 1, 2, 5, 6};
        for (int f : fusions) {
            for (int t : types) {
                PS p = base().set("fusion.typeFusion", f).set("pf.dameFusion", 25).detu(pet(t, 4));
                add(new Case("pet/fusion" + f + "/type" + t, p).op("calPoint").op("getDameAttack", "isMob", false, "seed", 11L));
            }
        }
        // pet with its own equipment: the pet recalculates first and its numbers feed the master
        PS rich = pet(5, 4).body(0, item(300, 0, o(77, 20), o(103, 10), o(50, 30), o(2, 5)));
        add(new Case("pet/pet-has-equipment", base().set("fusion.typeFusion", 8).set("pf.dameFusion", 10).detu(rich)).op("calPoint"));
        // a pet that is not fused still recalculates, the master does not add it
        add(new Case("pet/not-fused", base().set("fusion.typeFusion", 0).detu(pet(5, 1))).op("calPoint"));
        // gogeta: 1693 + 1553 in either order
        int[][] pairs = {{1693, 1553}, {1553, 1693}, {1693, 1693}, {1553, 1553}, {584, 1553}};
        for (int[] pr : pairs) {
            PS p = base().body(5, item(pr[0], 0)).detu(pet(0, 1).body(5, item(pr[1], 0)));
            add(new Case("pet/gogeta/" + pr[0] + "-" + pr[1], p).op("calPoint"));
        }
        add(new Case("pet/gogeta/pet-skin-empty", base().body(5, item(1693, 0)).detu(pet(0, 1))).op("calPoint"));
        // skins in slot 5
        for (int id : new int[]{584, 1207, 1208, 1209, 1210, 1211, 1233, 1234, 1235, 1236, 1237}) {
            add(new Case("skin/" + id, base().body(5, item(id, 0))).op("calPoint"));
        }
        // pet that is the main character: master supplies food and charm
        PS master = new PS("player").set("p.id", 1L).set("p.isPlayer", true).set("npoint.hpMax", 1000L).set("it.isEatMeal", true).set("npoint.tlTNSMPet", 15).set("ch.tdDeTu", ACTIVE);
        PS petMain = pet(2, 1).master(master);
        add(new Case("pet/as-main/eat-meal", petMain).op("calPoint").op("getDameAttack", "isMob", true, "seed", 12L).op("getDameAttack", "isMob", false, "seed", 13L));
        // clone (phan than)
        PS cloneMaster = new PS("player").set("p.id", 1L).set("p.isPlayer", true).set("npoint.hpMax", 900_000L).set("npoint.mpMax", 800_000L).set("npoint.dame", 70_000L);
        for (int pt : new int[]{-1, 1, 3, 7}) {
            PS clone = new PS("phanthan").set("p.id", 3L).set("p.isPhanThan", true).master(cloneMaster)
                    .set("npoint.hpg", 10L).set("npoint.mpg", 10L).set("npoint.dameg", 10L);
            if (pt >= 0) {
                clone.set("skill.id", 0).set("skill.damage", 100).set("skill.point", pt);
            }
            add(new Case("clone/skillpoint" + pt, clone).op("calPoint"));
        }
    }

    // ---------------------------------------------------------------- Whis ranking

    static void whisRanks() {
        for (int size : new int[]{2, 3, 4, 5, 8, 10, 12}) {
            for (int rank = 1; rank <= size + 1; rank++) {
                if (size == 12 && rank != 1 && rank != 3 && rank != 6 && rank != 10 && rank != 11 && rank != 13) {
                    continue;
                }
                PS p = base().set("p.id", 500L + rank);
                Case c = new Case("whis/size" + size + "/rank" + rank, p);
                long[] ids = new long[size];
                for (int i = 0; i < size; i++) {
                    ids[i] = 500L + (i + 1);
                }
                c.whis(ids);
                c.op("calPoint").op("getDameAttack", "isMob", false, "seed", 21L);
                add(c);
            }
        }
        PS notPl = base().set("p.isPlayer", false).set("p.isBoss", true).set("p.id", 501L);
        Case c = new Case("whis/not-player", notPl).whis(501, 502, 503, 504, 505, 506).op("calPoint");
        add(c);
    }

    @SuppressWarnings("unchecked")
    static PS trimBody(PS p, int size) {
        List<Object> body = (List<Object>) p.m.get("body");
        while (body.size() > size) {
            body.remove(body.size() - 1);
        }
        return p;
    }
