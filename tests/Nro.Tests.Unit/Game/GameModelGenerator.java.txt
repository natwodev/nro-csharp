    static String q(String s) {
        if (s == null) {
            return "null";
        }
        StringBuilder sb = new StringBuilder("\"");
        for (int i = 0; i < s.length(); i++) {
            char c = s.charAt(i);
            if (c == '"') {
                sb.append("\\\"");
            } else if (c == '\\') {
                sb.append("\\\\");
            } else if (c < 0x20 || c > 0x7e) {
                sb.append(String.format("\\u%04x", (int) c));
            } else {
                sb.append(c);
            }
        }
        return sb.append('"').toString();
    }

    /** A piece of JSON that is already formatted (an inner array), written as it is. */
    static final class Raw {
        final String json;

        Raw(String json) {
            this.json = json;
        }

        @Override
        public String toString() {
            return json;
        }
    }

    static Raw j(String json) {
        return new Raw(json);
    }

    static String arr(Object... items) {
        StringBuilder sb = new StringBuilder("[");
        for (int i = 0; i < items.length; i++) {
            if (i > 0) {
                sb.append(',');
            }
            Object o = items[i];
            if (o instanceof String) {
                sb.append(q((String) o));
            } else if (o == null) {
                sb.append("null");
            } else {
                sb.append(o.toString());
            }
        }
        return sb.append(']').toString();
    }

    public static char removeAccent(char ch) {
        int index = Arrays.binarySearch(SOURCE_CHARACTERS, ch);
        if (index >= 0) {
            ch = DESTINATION_CHARACTERS[index];
        }
        return ch;
    }

    public static String removeAccent(String str) {
        StringBuilder sb = new StringBuilder(str);
        for (int i = 0; i < sb.length(); i++) {
            sb.setCharAt(i, removeAccent(sb.charAt(i)));
        }
        return sb.toString();
    }

    // copied from Utils/TimeUtil.java: getTimeAgo (unchanged)
    public static String getTimeAgo(int timeRemainS) {
        int num = 0;
        if (timeRemainS > 60) {
            num = timeRemainS / 60;
            timeRemainS %= 60;
        }
        int num2 = 0;
        if (num > 60) {
            num2 = num / 60;
            num %= 60;
        }
        int num3 = 0;
        if (num2 > 24) {
            num3 = num2 / 24;
            num2 %= 24;
        }

        String text = "";
        if (num3 > 0) {
            text += num3;
            text += " ngày";
            text = text + num2 + " giờ";
        } else if (num2 > 0) {
            text += num2;
            text += " giờ";
            text = text + num + " phút";
        } else {
            if (num == 0) {
                num = 1;
            }
            text += num;
            text += " phút";
        }
        return text;
    }

    // nro/skill/NewSkill.java setSkillSpecial(): only the arithmetic on the target, fields become locals; returns {_xObjTaget, _yObjTaget}
    static int[] newSkillTarget(byte dir, int point, byte templateId, short _xPlayer, short _yPlayer, short _xObjTaget, short _yObjTaget) {
        short xObj;
        short yObj;
        int length = _xObjTaget - _xPlayer;
        int dx = dir * (point * 100 + 100);
        if (templateId != Skill.MA_PHONG_BA) {
            if (Math.abs(dx) < Math.abs(length) || Math.abs(length) < 100) {
                xObj = (short) dx;
            } else {
                xObj = (short) length;
            }
        } else {
            xObj = 75;
        }
        xObj = (short) Math.abs(xObj);
        yObj = (short) Math.abs(_yObjTaget);
        return new int[]{xObj, yObj};
    }

    // nro/power/CaptionManager.java getLevel(): the loop, with the powers as an array
    static int captionLevel(long[] powers, long power) {
        int size = powers.length;
        int level = 0;
        for (int i = size - 1; i >= 0; i--) {
            long p = powers[i];
            if (power >= p) {
                level = i;
                break;
            }
        }
        return level;
    }

    // nro/server/Manager.java: "String[] mc1 = rs.getString(..).split("-"); sideTask.count[0][0] = Integer.parseInt(mc1[0]); sideTask.count[0][1] = Integer.parseInt(mc1[1]);"
    static String parseRange(String text) {
        try {
            String[] mc = text.split("-");
            int a = Integer.parseInt(mc[0]);
            int b = Integer.parseInt(mc[1]);
            return arr("ok", a, b);
        } catch (NumberFormatException e) {
            return arr("error", "NumberFormatException", 0);
        } catch (ArrayIndexOutOfBoundsException e) {
            return arr("error", "ArrayIndexOutOfBoundsException", 0);
        }
    }
