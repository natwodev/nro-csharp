public class InventoryServiceGenerator {

    static class InvLogic {

    // ---------------------------------------------------------------- scenario helpers
    static final int[] UNIVERSE = {1, 2, 3, 4, 5, 6, 7, 8, 457, 1705};
    static final int[] QUANTITIES = {1, 2, 5, 9990, 9998, 9999, 10000, 50000, 99_990, 99_999_999, 100_000, 100_000_000, 2_000_000_000};
    static final int[] OPTION_IDS = {1, 4, 30, 31, 73, 93, 127, 128, 139, 140, 241, 242, 243, 244};
    static final int[] PARAMS = {0, 1, 5, 100, 29_999, 30_000, -3};

    static Random rnd;

    static ItemTemplate template(int id) {
        ItemTemplate t = new ItemTemplate();
        t.id = (short) id;
        t.type = 99;
        t.name = "item" + id;
        t.strRequire = id * 10;
        t.isUpToUp = (id >= 1 && id <= 4) || id == 1705;
        return t;
    }

    static Item item(int id, int quantity, int[][] options) {
        Item it = new Item();
        it.template = template(id);
        it.quantity = quantity;
        for (int[] o : options) it.itemOptions.add(new ItemOption(o[0], o[1]));
        return it;
    }

    static Item randomItem(boolean allowEmpty) {
        if (allowEmpty && rnd.nextInt(3) == 0) return new Item();
        int id = UNIVERSE[rnd.nextInt(UNIVERSE.length)];
        int q = QUANTITIES[rnd.nextInt(QUANTITIES.length)];
        int n = rnd.nextInt(4) == 0 ? 0 : 1 + rnd.nextInt(2);
        int[][] opts = new int[n][];
        for (int i = 0; i < n; i++) opts[i] = new int[]{OPTION_IDS[rnd.nextInt(OPTION_IDS.length)], PARAMS[rnd.nextInt(PARAMS.length)]};
        return item(id, q, opts);
    }

    static List<Item> randomList(int size, boolean allowEmpty) {
        List<Item> l = new ArrayList<>();
        for (int i = 0; i < size; i++) l.add(randomItem(allowEmpty));
        return l;
    }

    static String js(Item it) {
        if (it == null) return "null";
        if (!it.isNotNullItem()) return "0";
        StringBuilder sb = new StringBuilder("[").append(it.template.id).append(',').append(it.quantity).append(",[");
        for (int i = 0; i < it.itemOptions.size(); i++) {
            ItemOption o = it.itemOptions.get(i);
            sb.append(i > 0 ? "," : "").append('[').append(o.optionTemplate.id).append(',').append(o.param).append(']');
        }
        return sb.append("]]").toString();
    }

    static String js(List<Item> l) {
        StringBuilder sb = new StringBuilder("[");
        for (int i = 0; i < l.size(); i++) sb.append(i > 0 ? "," : "").append(js(l.get(i)));
        return sb.append("]").toString();
    }

    static String hex(byte[] b) {
        StringBuilder sb = new StringBuilder();
        for (byte x : b) sb.append(String.format("%02x", x));
        return sb.toString();
    }

    static Player player(int body, int bag, int box) {
        Player p = new Player();
        p.inventory = new Inventory();
        for (int i = 0; i < body; i++) p.inventory.itemsBody.add(new Item());
        for (int i = 0; i < bag; i++) p.inventory.itemsBag.add(new Item());
        for (int i = 0; i < box; i++) p.inventory.itemsBox.add(new Item());
        return p;
    }

    static final int[] SET_MARKERS = {127, 128, 129, 130, 131, 132, 133, 134, 135, 233, 250, 251, 252, 263, 265, 267};
    static final int[] SET_BONUS = {139, 140, 141, 142, 143, 144, 136, 137, 138, 234, 253, 254, 255, 264, 266, 268};

    static Item setItem(int idx, int marker, int[] bonuses, int bonusParam) {
        List<int[]> o = new ArrayList<>();
        if (marker >= 0) o.add(new int[]{marker, 1});
        for (int b : bonuses) o.add(new int[]{b, bonusParam});
        return item(10 + idx, 1, o.toArray(new int[0][]));
    }

    public static void main(String[] args) throws Exception {
        StringBuilder out = new StringBuilder();
        out.append("{\n  \"source\": \"nro/inventory/InventoryService.java (method bodies extracted verbatim)\",\n  \"sha256\": \"").append(args.length > 1 ? args[1] : "").append("\",\n");

        }
        out.append("  ],\n");

        int[] sizes = {0, 1, 100, 127, 128, 129, 200, 255, 256, 257};

            Item target = pick < list.size() ? list.get(pick) : pick == list.size() ? item(1, 1, new int[0][]) : null;
